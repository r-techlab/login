// POST endpoint for secure login authentication
function doPost(e) {
  try {
    // Try to parse as JSON first, fall back to URL-encoded form data
    var requestData;
    try {
      requestData = JSON.parse(e.postData.contents);
    } catch (jsonError) {
      // If JSON parsing fails, use e.parameter (URL-encoded form data)
      requestData = {};
      for (var key in e.parameter) {
        if (e.parameter.hasOwnProperty(key) && key !== 'callback') {
          requestData[key] = e.parameter[key];
        }
      }
    }
    
    var action = requestData.action || "login";
    
    if (action === "login") {
      return handleLogin(requestData, e);
    } else if (action === "validateSession") {
      return handleValidateSession(requestData, e);
    } else if (action === "logout") {
      return handleLogout(requestData, e);
    } else if (action === "uploadLogo") {
      return handleUploadLogoPost(requestData, e);
    } else if (action === "createLogo") {
      return handleCreateLogoPost(requestData, e);
    } else if (action === "updateLogo") {
      return handleUpdateLogoPost(requestData, e);
    } else {
      return createJsonResponse({ 
        status: "error", 
        message: "Invalid action" 
      });
    }
  } catch (error) {
    return createJsonResponse({ 
      status: "error", 
      message: "Server error: " + error.toString()
    });
  }
}

// LOGIN HANDLER
// ============================================

function handleLogin(requestData, e) {
  var loginid = requestData.loginid;
  var password = requestData.password;
  var ipAddress = getClientIP(e);
  
  // Validate input
  if (!loginid || !password) {
    logLoginAttempt(ipAddress, loginid, false, "Missing credentials");
    return createJsonResponse({ 
      status: "error", 
      message: "Missing credentials" 
    });
  }
  
  var ss = SpreadsheetApp.openById(SHEET_ID);
  
  // Check rate limiting
  var rateLimitResult = checkRateLimit(ss, ipAddress, loginid);
  if (!rateLimitResult.allowed) {
    logLoginAttempt(ipAddress, loginid, false, rateLimitResult.message);
    var result = { status: "error", message: rateLimitResult.message };
    if (callback) {
      return ContentService.createTextOutput(callback + "(" + JSON.stringify(result) + ")")
        .setMimeType(ContentService.MimeType.JAVASCRIPT);
    }
    return createJsonResponse(result);
  }
  
  var loginSheet = ss.getSheetByName("LogIn");
  var loginData = loginSheet.getDataRange().getValues();
  
  for (var i = 1; i < loginData.length; i++) {
    if (loginData[i][0] == loginid && loginData[i][2] == password) {
      var roleId = loginData[i][3];
      var userName = loginData[i][1];
      
      // Get role name
      var rolesSheet = ss.getSheetByName("Roles");
      var rolesData = rolesSheet.getDataRange().getValues();
      var roleName = "";
      
      for (var j = 1; j < rolesData.length; j++) {
        if (rolesData[j][0] == roleId) {
          roleName = rolesData[j][1];
          break;
        }
      }
      
      // Get menu access
      var menuAccess = getRoleMenuAccess(ss, roleId);
      
      // Create session
      var sessionId = generateSessionId();
      var now = new Date().getTime();
      var expiryTime = now + SESSION_DURATION;
      
      // Store session in ActiveSessions tab
      var sessionsSheet = ss.getSheetByName("ActiveSessions");
      sessionsSheet.appendRow([
        sessionId,
        loginid,
        userName,
        roleId,
        new Date(now),
        new Date(expiryTime),
        ipAddress,
        new Date(now)
      ]);
      
      // Log successful login
      logLoginAttempt(ipAddress, loginid, true, "Login successful");
      
      // Return session data
      return createJsonResponse({ 
        status: "success", 
        sessionId: sessionId,
        userId: loginid,
        userName: userName,
        roleId: roleId,
        roleName: roleName,
        menuAccess: menuAccess,
        expiryTime: expiryTime
      });
    }
  }
  
  // Invalid credentials
  logLoginAttempt(ipAddress, loginid, false, "Invalid credentials");
  return createJsonResponse({ 
    status: "error", 
    message: "Invalid credentials" 
  });
}

function handleLogout(requestData, e) {

      status: "error", 
      message: "Missing session ID" 
    });
  }
  
  var ss = SpreadsheetApp.openById(SHEET_ID);
  var sessionsSheet = ss.getSheetByName("ActiveSessions");
  var data = sessionsSheet.getDataRange().getValues();
  
  // Find and delete session

      sessionsSheet.deleteRow(i + 1);
      return createJsonResponse({ 
        status: "success", 
        message: "Logged out successfully" 
      });
    }
  }
  
  return createJsonResponse({ 
    status: "success", 
    message: "Session not found or already expired" 
  });
}

// Validate session
function validateSession(ss, sessionId, userId, ipAddress) {
  var sessionsSheet = ss.getSheetByName("ActiveSessions");
  var data = sessionsSheet.getDataRange().getValues();
  var now = new Date().getTime();
  
  for (var i = 1; i < data.length; i++) {
    if (data[i][0] == sessionId && data[i][1] == userId) {
      var expiryTime = new Date(data[i][5]).getTime();
      var sessionIP = data[i][6];
      
      // Check if expired
      if (now > expiryTime) {
        sessionsSheet.deleteRow(i + 1);
        return { 
          status: "error",
          valid: false, 
          message: "Session expired" 
        };
      }
      
      // Check IP address match
      if (sessionIP !== ipAddress) {
        return { 
          status: "error",
          valid: false, 
          message: "IP address mismatch - security violation" 
        };
      }
      
      // Update last activity
      sessionsSheet.getRange(i + 1, 8).setValue(new Date());
      
      // Get user data
      var userName = data[i][2];
      var roleId = data[i][3];
      
      // Get role name
      var rolesSheet = ss.getSheetByName("Roles");
      var rolesData = rolesSheet.getDataRange().getValues();
      var roleName = "";
      
      for (var j = 1; j < rolesData.length; j++) {
        if (rolesData[j][0] == roleId) {
          roleName = rolesData[j][1];
          break;
        }
      }
      
      // Get menu access
      var menuAccess = getRoleMenuAccess(ss, roleId);
      
      return { 
        status: "success",
        valid: true,
        userId: userId,
        userName: userName,
        roleId: roleId,
        roleName: roleName,
        menuAccess: menuAccess,
        expiryTime: expiryTime
      };
    }
  }
  
  return { 
    status: "error",
    valid: false, 
    message: "Invalid session" 
  };
}

// Check rate limiting
function checkRateLimit(ss, ipAddress, userId) {
  var attemptsSheet = ss.getSheetByName("LoginAttempts");
  var data = attemptsSheet.getDataRange().getValues();
  var now = new Date().getTime();
  var windowStart = now - ATTEMPT_WINDOW;
  
  var ipAttempts = 0;
  var userAttempts = 0;
  
  for (var i = 1; i < data.length; i++) {
    var attemptTime = new Date(data[i][0]).getTime();
    if (attemptTime > windowStart) {
      var attemptIP = data[i][1];
      var attemptUser = data[i][2];
      var success = data[i][3];
      
      if (!success) {
        if (attemptIP === ipAddress) {
          ipAttempts++;
        }
        if (attemptUser === userId) {
          userAttempts++;
        }
      }
    }
  }
  
  if (ipAttempts >= MAX_LOGIN_ATTEMPTS) {
    return { 
      allowed: false, 
      message: "Too many failed attempts from this IP. Try again in 15 minutes." 
    };
  }
  
  if (userAttempts >= MAX_LOGIN_ATTEMPTS) {
    return { 
      allowed: false, 
      message: "Too many failed attempts for this user. Try again in 15 minutes." 
    };
  }
  
  return { allowed: true };
}

// Log login attempt
function logLoginAttempt(ipAddress, userId, success, message) {
  try {
    var ss = SpreadsheetApp.openById(SHEET_ID);
    var attemptsSheet = ss.getSheetByName("LoginAttempts");
    attemptsSheet.appendRow([

      ipAddress,
      userId,
      success,
      message
    ]);
  } catch (error) {
    // Fail silently if logging fails
  }
}

// Generate secure session ID
function generateSessionId() {
  return 'sess_' + Utilities.getUuid();
}

// Get client IP address
function getClientIP(e) {
  try {
    // Try to get real IP from headers
    if (e.parameter && e.parameter['X-Forwarded-For']) {
      return e.parameter['X-Forwarded-For'].split(',')[0].trim();
    }
    // Fallback to a placeholder
    return 'unknown';
  } catch (error) {
    return 'unknown';
  }
}

// Get menu access for role
function getRoleMenuAccess(ss, roleId) {
  var roleMenuSheet = ss.getSheetByName("RoleMenuAccess");
  var roleMenuData = roleMenuSheet.getDataRange().getValues();
  
  var menuMasterSheet = ss.getSheetByName("MenuMaster");
  var menuMasterData = menuMasterSheet.getDataRange().getValues();
  
  var allowedMenuIds = [];
  for (var i = 1; i < roleMenuData.length; i++) {
    if (roleMenuData[i][0] == roleId) {
      allowedMenuIds.push(roleMenuData[i][1]);
    }
  }
  
  var menus = [];
  for (var j = 1; j < menuMasterData.length; j++) {
    var menuId = menuMasterData[j][0];
    
    if (allowedMenuIds.indexOf(menuId) !== -1) {
      menus.push({
        menuId: menuId,
        parentMenuId: menuMasterData[j][1],
        menuType: menuMasterData[j][2],
        menuName: menuMasterData[j][3],
        pageUrl: menuMasterData[j][4],
        sortOrder: menuMasterData[j][5]
      });
    }
  }
  
  menus.sort(function(a, b) {
    return a.sortOrder - b.sortOrder;
  });
  
  return menus;
}

// Get all users
function handleGetUsersGet(e) {
  var sessionId = e.parameter.sessionId;
  var userId = e.parameter.userId;
  var callback = e.parameter.callback;
  var ipAddress = getClientIP(e);
  
  // Validate session
  var ss = SpreadsheetApp.openById(SHEET_ID);
  var validation = validateSession(ss, sessionId, userId, ipAddress);
  
  if (!validation.valid) {
    var errorResult = { status: "error", message: "Invalid or expired session" };
    if (callback) {
      return ContentService.createTextOutput(callback + "(" + JSON.stringify(errorResult) + ")")
        .setMimeType(ContentService.MimeType.JAVASCRIPT);
    }
    return createJsonResponse(errorResult);
  }
  
  // Check if user is admin
  if (validation.roleName !== "Admin") {
    var errorResult = { status: "error", message: "Access denied: Admin role required" };
    if (callback) {
      return ContentService.createTextOutput(callback + "(" + JSON.stringify(errorResult) + ")")
        .setMimeType(ContentService.MimeType.JAVASCRIPT);
    }
    return createJsonResponse(errorResult);
  }
  
  // Get all users from LogIn sheet
  var loginSheet = ss.getSheetByName("LogIn");
  var loginData = loginSheet.getDataRange().getValues();
  var users = [];
  
  for (var i = 1; i < loginData.length; i++) {
    users.push({
      loginId: loginData[i][0],
      userName: loginData[i][1],
      password: loginData[i][2],
      roleId: loginData[i][3]
    });
  }
  
  var result = { status: "success", users: users };
  if (callback) {
    return ContentService.createTextOutput(callback + "(" + JSON.stringify(result) + ")")
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return createJsonResponse(result);
}

  var sortOrder = e.parameter.sortOrder;
  var callback = e.parameter.callback;
  var ipAddress = getClientIP(e);

        .setMimeType(ContentService.MimeType.JAVASCRIPT);

  // Check if user is admin
  if (validation.roleName !== "Admin") {
    var errorResult = { status: "error", message: "Access denied: Admin role required" };
    if (callback) {
      return ContentService.createTextOutput(callback + "(" + JSON.stringify(errorResult) + ")")
        .setMimeType(ContentService.MimeType.JAVASCRIPT);

      return ContentService.createTextOutput(callback + "(" + JSON.stringify(errorResult) + ")")

    if (menuMasterData[i][0] > maxId) {
      maxId = menuMasterData[i][0];
    }

  var newMenuId = maxId + 1;
  
  // Add new menu
  menuMasterSheet.appendRow([
    newMenuId,
    parentMenuId === 'NULL' ? '' : parentMenuId,
    menuType,
    menuName,
    pageUrl === 'NULL' ? '' : pageUrl,
    sortOrder
  ]);
  
  var result = { status: "success", message: "Menu created successfully", menuId: newMenuId };
  if (callback) {
    return ContentService.createTextOutput(callback + "(" + JSON.stringify(result) + ")")
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return createJsonResponse(result);
}

// Get all sales
function handleGetSalesGet(e) {
  var sessionId = e.parameter.sessionId;
  var userId = e.parameter.userId;
  var callback = e.parameter.callback;
  var ipAddress = getClientIP(e);
  
  // Validate session
  var ss = SpreadsheetApp.openById(SHEET_ID);
  var validation = validateSession(ss, sessionId, userId, ipAddress);
  
  if (!validation.valid) {
    var errorResult = { status: "error", message: "Invalid or expired session" };
    if (callback) {
      return ContentService.createTextOutput(callback + "(" + JSON.stringify(errorResult) + ")")
        .setMimeType(ContentService.MimeType.JAVASCRIPT);
    }
    return createJsonResponse(errorResult);
  }
  
  // Read pagination and search parameters
  var page = parseInt(e.parameter.page) || 1;
  var pageSize = parseInt(e.parameter.pageSize) || 10;
  var search = (e.parameter.search || '').toLowerCase();
  
  // Get all sales from SalesHeader sheet
  var salesHeaderSheet = ss.getSheetByName("SalesHeader");
  var salesHeaderData = salesHeaderSheet.getDataRange().getValues();
  var allSales = [];
  
  for (var i = 1; i < salesHeaderData.length; i++) {
    if (salesHeaderData[i][1]) { // Check if DocNo exists
      var docNo = salesHeaderData[i][1].toString();
      var customer = salesHeaderData[i][3] || '';
      var narration = salesHeaderData[i][4] || '';
      var date = salesHeaderData[i][2] ? Utilities.formatDate(new Date(salesHeaderData[i][2]), Session.getScriptTimeZone(), 'yyyy-MM-dd') : '';
      
      // Apply search filter if search term is provided
      if (search) {
        if (docNo.toLowerCase().indexOf(search) === -1 &&
            customer.toLowerCase().indexOf(search) === -1 &&
            narration.toLowerCase().indexOf(search) === -1 &&
            date.toLowerCase().indexOf(search) === -1) {
          continue; // Skip this record if no match
        }
      }
      
      allSales.push({
        docType: salesHeaderData[i][0] || '',
        docNo: docNo,
        date: date,
        customer: customer,
        salesman: salesHeaderData[i][18] || '',
        narration: narration,
        grossAmount: salesHeaderData[i][5] || 0,
        discountPercentage: salesHeaderData[i][6] || 0,
        discountAmount: salesHeaderData[i][7] || 0,
        vat: salesHeaderData[i][8] || 0,
        net: salesHeaderData[i][9] || 0,
        createdBy: salesHeaderData[i][10] || '',
        createdTime: salesHeaderData[i][11] || '',
        address: salesHeaderData[i][12] || '',
        trn: salesHeaderData[i][13] || '',
        vatTreatment: salesHeaderData[i][14] || '',
        rateType: salesHeaderData[i][15] || '',
        refNo: salesHeaderData[i][16] || '',
        refDate: salesHeaderData[i][17] ? Utilities.formatDate(new Date(salesHeaderData[i][17]), Session.getScriptTimeZone(), 'yyyy-MM-dd') : ''
      });
    }
  }
  
  // Sort descending by DocNo (most recent/last document first)
  allSales.sort(function(a, b) {
    return b.docNo.localeCompare(a.docNo);
  });
  
  var totalRecords = allSales.length;
  var totalPages = Math.ceil(totalRecords / pageSize);
  
  // Get only the records for the requested page
  var startIndex = (page - 1) * pageSize;
  var endIndex = Math.min(startIndex + pageSize, totalRecords);
  var sales = allSales.slice(startIndex, endIndex);
  
  var result = { 
    status: "success", 
    sales: sales, 
    totalRecords: totalRecords,
    totalPages: totalPages,
    currentPage: page,
    pageSize: pageSize
  };
  if (callback) {
    return ContentService.createTextOutput(callback + "(" + JSON.stringify(result) + ")")
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return createJsonResponse(result);
}

// Create new sale
function handleCreateSaleGet(e) {
  var sessionId = e.parameter.sessionId;
  var userId = e.parameter.userId;
  var saleDataJson = e.parameter.saleData;
  var callback = e.parameter.callback;
  var ipAddress = getClientIP(e);
  
  // Validate session
  var ss = SpreadsheetApp.openById(SHEET_ID);
  var validation = validateSession(ss, sessionId, userId, ipAddress);
  
  if (!validation.valid) {
    var errorResult = { status: "error", message: "Invalid or expired session" };
    if (callback) {
      return ContentService.createTextOutput(callback + "(" + JSON.stringify(errorResult) + ")")
        .setMimeType(ContentService.MimeType.JAVASCRIPT);
    }
    return createJsonResponse(errorResult);
  }
  
  // Parse sale data
  var saleData;
  try {
    saleData = JSON.parse(saleDataJson);
  } catch (error) {
    var errorResult = { status: "error", message: "Invalid sale data format" };
    if (callback) {
      return ContentService.createTextOutput(callback + "(" + JSON.stringify(errorResult) + ")")
        .setMimeType(ContentService.MimeType.JAVASCRIPT);
    }
    return createJsonResponse(errorResult);
  }
  
  // Validate input
  if (!saleData.header || !saleData.details || saleData.details.length === 0) {
    var errorResult = { status: "error", message: "Header and at least one detail line are required" };
    if (callback) {
      return ContentService.createTextOutput(callback + "(" + JSON.stringify(errorResult) + ")")
        .setMimeType(ContentService.MimeType.JAVASCRIPT);
    }
    return createJsonResponse(errorResult);
  }
  
  // Generate DocNo
  var salesHeaderSheet = ss.getSheetByName("SalesHeader");
  var salesHeaderData = salesHeaderSheet.getDataRange().getValues();
  var maxNum = 10000; // Starting number
  
  for (var i = 1; i < salesHeaderData.length; i++) {
    if (salesHeaderData[i][1]) {
      var docNo = salesHeaderData[i][1].toString();
      if (docNo.startsWith('SI')) {
        var num = parseInt(docNo.substring(2));
        if (num > maxNum) {
          maxNum = num;
        }
      }
    }
  }
  
  var newDocNo = 'SI' + (maxNum + 1);
  
  // Add header
  var header = saleData.header;
  var now = new Date();
  var createdTime = Utilities.formatDate(now, Session.getScriptTimeZone(), 'yyyy-MM-dd HH:mm:ss');
  
  salesHeaderSheet.appendRow([
    'SI', // DocType
    newDocNo,
    header.date,
    header.customer,
    header.narration || '',
    header.grossAmount || 0,
    header.discountPercentage || 0,
    header.discountAmount || 0,
    header.vat || 0,
    header.net || 0,
    validation.userName, // CreatedBy
    createdTime,
    header.address || '',
    header.trn || '',
    header.vatTreatment || '',
    header.rateType || '',
    header.refNo || '',
    header.refDate || '',
    header.salesman || ''
  ]);
  
  // Add details
  var salesDetailsSheet = ss.getSheetByName("SalesDetails");
  for (var i = 0; i < saleData.details.length; i++) {
    var detail = saleData.details[i];
    salesDetailsSheet.appendRow([
      'SI', // DocType
      newDocNo,
      detail.docSrNo || (i + 1),
      detail.stockCode,
      detail.stockDescription || '',
      detail.qty || 0,
      detail.unit || '',
      detail.rate || 0,
      detail.grossAmount || 0,
      detail.discountAmount || 0,
      detail.discountPercentage || 0,
      detail.vatAmount || 0,
      detail.vatPercentage || 0,
      detail.netAmount || 0,
      detail.vatTreatment || '',
      detail.rateType || ''
    ]);
  }
  
  var result = { status: "success", message: "Sale created successfully", docNo: newDocNo };
  if (callback) {
    return ContentService.createTextOutput(callback + "(" + JSON.stringify(result) + ")")
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return createJsonResponse(result);
}

// Get all purchases
function handleGetPurchasesGet(e) {
  var sessionId = e.parameter.sessionId;
  var userId = e.parameter.userId;
  var callback = e.parameter.callback;
  var ipAddress = getClientIP(e);
  
  // Validate session
  var ss = SpreadsheetApp.openById(SHEET_ID);
  var validation = validateSession(ss, sessionId, userId, ipAddress);
  
  if (!validation.valid) {
    var errorResult = { status: "error", message: "Invalid or expired session" };
    if (callback) {
      return ContentService.createTextOutput(callback + "(" + JSON.stringify(errorResult) + ")")
        .setMimeType(ContentService.MimeType.JAVASCRIPT);
    }
    return createJsonResponse(errorResult);
  }
  
  // Read pagination and search parameters
  var page = parseInt(e.parameter.page) || 1;
  var pageSize = parseInt(e.parameter.pageSize) || 10;
  var search = (e.parameter.search || '').toLowerCase();
  
  // Get all purchases from PurchaseHeader sheet
  var purchaseHeaderSheet = ss.getSheetByName("PurchaseHeader");
  var purchaseHeaderData = purchaseHeaderSheet.getDataRange().getValues();
  var allPurchases = [];
  
  for (var i = 1; i < purchaseHeaderData.length; i++) {
    if (purchaseHeaderData[i][1]) { // Check if DocNo exists
      var docNo = purchaseHeaderData[i][1].toString();
      var supplier = purchaseHeaderData[i][3] || '';
      var narration = purchaseHeaderData[i][4] || '';
      var date = purchaseHeaderData[i][2] ? Utilities.formatDate(new Date(purchaseHeaderData[i][2]), Session.getScriptTimeZone(), 'yyyy-MM-dd') : '';
      
      // Apply search filter if search term is provided
      if (search) {
        if (docNo.toLowerCase().indexOf(search) === -1 &&
            supplier.toLowerCase().indexOf(search) === -1 &&
            narration.toLowerCase().indexOf(search) === -1 &&
            date.toLowerCase().indexOf(search) === -1) {
          continue; // Skip this record if no match
        }
      }
      
      allPurchases.push({
        docType: purchaseHeaderData[i][0] || '',
        docNo: docNo,
        date: date,
        supplier: supplier,
        salesman: purchaseHeaderData[i][18] || '',
        narration: narration,
        grossAmount: purchaseHeaderData[i][5] || 0,
        discountPercentage: purchaseHeaderData[i][6] || 0,
        discountAmount: purchaseHeaderData[i][7] || 0,
        vat: purchaseHeaderData[i][8] || 0,
        net: purchaseHeaderData[i][9] || 0,
        createdBy: purchaseHeaderData[i][10] || '',
        createdTime: purchaseHeaderData[i][11] || '',
        address: purchaseHeaderData[i][12] || '',
        trn: purchaseHeaderData[i][13] || '',
        vatTreatment: purchaseHeaderData[i][14] || '',
        rateType: purchaseHeaderData[i][15] || '',
        refNo: purchaseHeaderData[i][16] || '',
        refDate: purchaseHeaderData[i][17] ? Utilities.formatDate(new Date(purchaseHeaderData[i][17]), Session.getScriptTimeZone(), 'yyyy-MM-dd') : ''
      });
    }
  }
  
  // Sort descending by DocNo (most recent/last document first)
  allPurchases.sort(function(a, b) {
    return b.docNo.localeCompare(a.docNo);
  });
  
  var totalRecords = allPurchases.length;
  var totalPages = Math.ceil(totalRecords / pageSize);
  
  // Get only the records for the requested page
  var startIndex = (page - 1) * pageSize;
  var endIndex = Math.min(startIndex + pageSize, totalRecords);
  var purchases = allPurchases.slice(startIndex, endIndex);
  
  var result = { 
    status: "success", 
    purchases: purchases, 
    totalRecords: totalRecords,
    totalPages: totalPages,
    currentPage: page,
    pageSize: pageSize
  };
  if (callback) {
    return ContentService.createTextOutput(callback + "(" + JSON.stringify(result) + ")")
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return createJsonResponse(result);
}

// Get sales report with date range and filters
function handleGetSalesReportGet(e) {
  var sessionId = e.parameter.sessionId;
  var userId = e.parameter.userId;
  var callback = e.parameter.callback;
  var ipAddress = getClientIP(e);
  
  // Validate session
  var ss = SpreadsheetApp.openById(SHEET_ID);
  var validation = validateSession(ss, sessionId, userId, ipAddress);
  
  if (!validation.valid) {
    var errorResult = { status: "error", message: "Invalid or expired session" };
    if (callback) {
      return ContentService.createTextOutput(callback + "(" + JSON.stringify(errorResult) + ")")
        .setMimeType(ContentService.MimeType.JAVASCRIPT);
    }
    return createJsonResponse(errorResult);
  }
  
  // Read filter parameters
  var fromDate = e.parameter.fromDate || '';
  var toDate = e.parameter.toDate || '';
  var customerFilter = (e.parameter.customer || '').toLowerCase();
  var salesmanFilter = (e.parameter.salesman || '').toLowerCase();
  var reportType = (e.parameter.reportType || 'headerwise').toLowerCase();
  
  // Build customer lookup map (code -> description)
  var customerSheet = ss.getSheetByName("CustomerMaster");
  var customerData = customerSheet.getDataRange().getValues();
  var customerMap = {};
  for (var c = 1; c < customerData.length; c++) {
    if (customerData[c][0]) {
      customerMap[String(customerData[c][0]).toLowerCase()] = customerData[c][1] || '';
    }
  }
  
  // Build salesman lookup map (code -> description)
  var salesmanSheet = ss.getSheetByName("Salesman");
  var salesmanData = salesmanSheet.getDataRange().getValues();
  var salesmanMap = {};
  for (var s = 1; s < salesmanData.length; s++) {
    if (salesmanData[s][0]) {
      salesmanMap[String(salesmanData[s][0]).toLowerCase()] = salesmanData[s][1] || '';
    }
  }
  
  // Get all sales from SalesHeader sheet
  var salesHeaderSheet = ss.getSheetByName("SalesHeader");
  var salesHeaderData = salesHeaderSheet.getDataRange().getValues();
  var allSales = [];
  
  for (var i = 1; i < salesHeaderData.length; i++) {
    if (salesHeaderData[i][1]) { // Check if DocNo exists
      var docNo = salesHeaderData[i][1].toString();
      var customer = salesHeaderData[i][3] || '';
      var salesman = String(salesHeaderData[i][18] || '');
      var dateObj = salesHeaderData[i][2] ? new Date(salesHeaderData[i][2]) : null;
      var dateStr = dateObj ? Utilities.formatDate(dateObj, Session.getScriptTimeZone(), 'yyyy-MM-dd') : '';
      
      // Apply date range filter
      if (fromDate && dateStr < fromDate) continue;
      if (toDate && dateStr > toDate) continue;
      
      // Apply customer filter
      if (customerFilter && customer.toLowerCase().indexOf(customerFilter) === -1) continue;
      
      // Apply salesman filter
      if (salesmanFilter && salesman.toLowerCase().indexOf(salesmanFilter) === -1) continue;
      
      // Look up customer and salesman names
      var customerName = customerMap[customer.toLowerCase()] || '';
      var salesmanName = salesmanMap[salesman.toLowerCase()] || '';
      
      allSales.push({
        docType: salesHeaderData[i][0] || '',
        docNo: docNo,
        date: dateStr,
        customer: customer,
        customerName: customerName,
        salesman: salesman,
        salesmanName: salesmanName,
        narration: salesHeaderData[i][4] || '',
        grossAmount: parseFloat(salesHeaderData[i][5]) || 0,
        discountPercentage: parseFloat(salesHeaderData[i][6]) || 0,
        discountAmount: parseFloat(salesHeaderData[i][7]) || 0,
        vat: parseFloat(salesHeaderData[i][8]) || 0,
        net: parseFloat(salesHeaderData[i][9]) || 0,
        vatTreatment: salesHeaderData[i][14] || '',
        rateType: salesHeaderData[i][15] || ''
      });
    }
  }
  
  // Sort by date descending, then by DocNo descending
  allSales.sort(function(a, b) {
    if (a.date !== b.date) return b.date.localeCompare(a.date);
    return b.docNo.localeCompare(a.docNo);
  });
  
  // If reportType is itemwise, fetch item details for each sale
  if (reportType === 'itemwise') {
    var salesDetailsSheet = ss.getSheetByName("SalesDetails");
    var salesDetailsData = salesDetailsSheet.getDataRange().getValues();
    
    // Build a map of docNo -> details array
    var detailsMap = {};
    for (var d = 1; d < salesDetailsData.length; d++) {
      var detailDocNo = salesDetailsData[d][1];
      if (detailDocNo) {
        if (!detailsMap[detailDocNo]) {
          detailsMap[detailDocNo] = [];
        }
        detailsMap[detailDocNo].push({
          docSrNo: salesDetailsData[d][2] || 0,
          stockCode: salesDetailsData[d][3] || '',
          stockDescription: salesDetailsData[d][4] || '',
          qty: parseFloat(salesDetailsData[d][5]) || 0,
          unit: salesDetailsData[d][6] || '',
          rate: parseFloat(salesDetailsData[d][7]) || 0,
          grossAmount: parseFloat(salesDetailsData[d][8]) || 0,
          discountAmount: parseFloat(salesDetailsData[d][9]) || 0,
          discountPercentage: parseFloat(salesDetailsData[d][10]) || 0,
          vatAmount: parseFloat(salesDetailsData[d][11]) || 0,
          vatPercentage: parseFloat(salesDetailsData[d][12]) || 0,
          netAmount: parseFloat(salesDetailsData[d][13]) || 0,
          vatTreatment: salesDetailsData[d][14] || '',
          rateType: salesDetailsData[d][15] || ''
        });
      }
    }
    
    // Attach details to each sale
    for (var si = 0; si < allSales.length; si++) {
      allSales[si].details = detailsMap[allSales[si].docNo] || [];
    }
  }
  
  // Calculate summary
  var totalCount = allSales.length;
  var totalGross = 0;
  var totalDiscount = 0;
  var totalVat = 0;
  var totalNet = 0;
  var totalItemQty = 0;
  
  for (var i = 0; i < allSales.length; i++) {
    totalGross += allSales[i].grossAmount;
    totalDiscount += allSales[i].discountAmount;
    totalVat += allSales[i].vat;
    totalNet += allSales[i].net;
    if (reportType === 'itemwise' && allSales[i].details) {
      for (var di = 0; di < allSales[i].details.length; di++) {
        totalItemQty += allSales[i].details[di].qty;
      }
    }
  }
  
  var result = { 
    status: "success", 
    sales: allSales,
    reportType: reportType,
    summary: {
      totalCount: totalCount,
      totalGross: Math.round(totalGross * 100) / 100,
      totalDiscount: Math.round(totalDiscount * 100) / 100,
      totalVat: Math.round(totalVat * 100) / 100,
      totalNet: Math.round(totalNet * 100) / 100,
      totalItemQty: Math.round(totalItemQty * 100) / 100
    }
  };
  
  if (callback) {
    return ContentService.createTextOutput(callback + "(" + JSON.stringify(result) + ")")
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return createJsonResponse(result);
}

function cleanupExpiredSessions() {
  var ss = SpreadsheetApp.openById(SHEET_ID);
  var sessionsSheet = ss.getSheetByName("ActiveSessions");
  var data = sessionsSheet.getDataRange().getValues();
  var now = new Date().getTime();
  
  // Delete from bottom to top to avoid index issues
  for (var i = data.length - 1; i >= 1; i--) {
    var expiryTime = new Date(data[i][5]).getTime();
    if (now > expiryTime) {
      sessionsSheet.deleteRow(i + 1);
    }
  }
  
  Logger.log("Cleaned up expired sessions");
}

    if (dashboardData[i][0]) { // Check if Code exists
      var rolesStr = dashboardData[i][2] || '';
      var usersStr = dashboardData[i][3] || '';
      
      // Parse roles XML
      var roleMatch = false;
      if (rolesStr) {
        var roleRegex = new RegExp('<role>([^<]*)<\\/role>', 'gi');
        var roleMatchResult;
        while ((roleMatchResult = roleRegex.exec(rolesStr)) !== null) {
          if (roleMatchResult[1].trim() == roleId) {
            roleMatch = true;
            break;
          }
        }
      }
      
      // Parse users XML
      var userMatch = false;
      if (usersStr) {
        var userRegex = new RegExp('<user>([^<]*)<\\/user>', 'gi');
        var userMatchResult;
        while ((userMatchResult = userRegex.exec(usersStr)) !== null) {
          if (userMatchResult[1].trim().toLowerCase() == userId.toLowerCase()) {
            userMatch = true;
            break;
          }
        }
      }
      
      // If both roles and users are empty, show to everyone
      // Otherwise, check if user's role or user ID matches
      var showDashboard = false;
      if (!rolesStr && !usersStr) {
        showDashboard = true; // Visible to all
      } else if (roleMatch || userMatch) {
        showDashboard = true;
      }
      
      if (showDashboard) {
        userDashboard.push({
          code: dashboardData[i][0].toString(),
          description: dashboardData[i][1] || ''
        });
      }
    }
  }
  
  var result = { status: "success", dashboard: userDashboard };
  if (callback) {
    return ContentService.createTextOutput(callback + "(" + JSON.stringify(result) + ")")
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return createJsonResponse(result);
}

// Update dashboard record
function handleUpdateDashboardGet(e) {
  var sessionId = e.parameter.sessionId;
  var userId = e.parameter.userId;
  var code = e.parameter.code;
  var description = e.parameter.description || '';
  var roles = e.parameter.roles || '';
  var users = e.parameter.users || '';
  var callback = e.parameter.callback;
  var ipAddress = getClientIP(e);
  
  // Validate session
  var ss = SpreadsheetApp.openById(SHEET_ID);
  var validation = validateSession(ss, sessionId, userId, ipAddress);
  
  if (!validation.valid) {
    var errorResult = { status: "error", message: "Invalid or expired session" };
    if (callback) {
      return ContentService.createTextOutput(callback + "(" + JSON.stringify(errorResult) + ")")
        .setMimeType(ContentService.MimeType.JAVASCRIPT);
    }
    return createJsonResponse(errorResult);
  }
  
  // Validate input
  if (!code) {
    var errorResult = { status: "error", message: "Code is required" };
    if (callback) {
      return ContentService.createTextOutput(callback + "(" + JSON.stringify(errorResult) + ")")
        .setMimeType(ContentService.MimeType.JAVASCRIPT);
    }
    return createJsonResponse(errorResult);
  }
  
  var dashboardSheet = ss.getSheetByName("Dashboard");
  if (!dashboardSheet) {
    dashboardSheet = ss.insertSheet("Dashboard");
    dashboardSheet.appendRow(["Code", "Description", "Roles", "Users"]);
  }
  
  var dashboardData = dashboardSheet.getDataRange().getValues();
  
  // Find and update record
  var found = false;
  for (var i = 1; i < dashboardData.length; i++) {
    if (dashboardData[i][0] && dashboardData[i][0].toString() == code.toString()) {
      dashboardSheet.getRange(i + 1, 1, 1, 4).setValues([[code, description, roles, users]]);
      found = true;
      break;
    }
  }
  
  if (!found) {
    // If not found, append as new row
    dashboardSheet.appendRow([code, description, roles, users]);
  }
  
  var result = { status: "success", message: "Dashboard record updated successfully" };
  if (callback) {
    return ContentService.createTextOutput(callback + "(" + JSON.stringify(result) + ")")
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return createJsonResponse(result);
}

// Get sales summary grouped by salesman for the current month
function handleGetSalesmanSalesSummaryGet(e) {
  var sessionId = e.parameter.sessionId;
  var userId = e.parameter.userId;
  var callback = e.parameter.callback;
  var ipAddress = getClientIP(e);
  
  // Validate session
  var ss = SpreadsheetApp.openById(SHEET_ID);
  var validation = validateSession(ss, sessionId, userId, ipAddress);
  
  if (!validation.valid) {
    var errorResult = { status: "error", message: "Invalid or expired session" };
    if (callback) {
      return ContentService.createTextOutput(callback + "(" + JSON.stringify(errorResult) + ")")
        .setMimeType(ContentService.MimeType.JAVASCRIPT);
    }
    return createJsonResponse(errorResult);
  }
  
  // Calculate current month date range
  var now = new Date();
  var year = now.getFullYear();
  var month = now.getMonth(); // 0-based
  var firstDay = new Date(year, month, 1);
  var lastDay = new Date(year, month + 1, 0); // Last day of current month
  
  var fromDate = Utilities.formatDate(firstDay, Session.getScriptTimeZone(), 'yyyy-MM-dd');
  var toDate = Utilities.formatDate(lastDay, Session.getScriptTimeZone(), 'yyyy-MM-dd');
  
  // Build salesman lookup map (code -> description)
  var salesmanSheet = ss.getSheetByName("Salesman");
  var salesmanData = salesmanSheet.getDataRange().getValues();
  var salesmanMap = {};
  for (var s = 1; s < salesmanData.length; s++) {
    if (salesmanData[s][0]) {
      salesmanMap[String(salesmanData[s][0]).toLowerCase()] = salesmanData[s][1] || '';
    }
  }
  
  // Get all sales from SalesHeader sheet
  var salesHeaderSheet = ss.getSheetByName("SalesHeader");
  var salesHeaderData = salesHeaderSheet.getDataRange().getValues();
  
  // Aggregate sales by salesman
  var salesBySalesman = {};
  
  for (var i = 1; i < salesHeaderData.length; i++) {
    if (salesHeaderData[i][1]) { // Check if DocNo exists
      var dateObj = salesHeaderData[i][2] ? new Date(salesHeaderData[i][2]) : null;
      var dateStr = dateObj ? Utilities.formatDate(dateObj, Session.getScriptTimeZone(), 'yyyy-MM-dd') : '';
      
      // Filter by current month
      if (dateStr < fromDate || dateStr > toDate) continue;
      
      var salesmanCode = String(salesHeaderData[i][18] || '').toLowerCase();
      if (!salesmanCode) continue;
      
      var grossAmount = parseFloat(salesHeaderData[i][5]) || 0;
      var netAmount = parseFloat(salesHeaderData[i][9]) || 0;
      
      if (!salesBySalesman[salesmanCode]) {
        salesBySalesman[salesmanCode] = {
          salesmanCode: salesmanCode,
          salesmanName: salesmanMap[salesmanCode] || salesmanCode,
          totalGross: 0,
          totalNet: 0,
          invoiceCount: 0
        };
      }
      
      salesBySalesman[salesmanCode].totalGross += grossAmount;
      salesBySalesman[salesmanCode].totalNet += netAmount;
      salesBySalesman[salesmanCode].invoiceCount++;
    }
  }
  
  // Convert to array and sort by totalGross descending
  var summary = [];
  for (var key in salesBySalesman) {
    if (salesBySalesman.hasOwnProperty(key)) {
      var item = salesBySalesman[key];
      item.totalGross = Math.round(item.totalGross * 100) / 100;
      item.totalNet = Math.round(item.totalNet * 100) / 100;
      summary.push(item);
    }
  }
  
  summary.sort(function(a, b) {
    return b.totalGross - a.totalGross;
  });
  
  var result = { 
    status: "success", 
    summary: summary,
    month: month + 1,
    year: year,
    fromDate: fromDate,
    toDate: toDate
  };
  
  if (callback) {
    return ContentService.createTextOutput(callback + "(" + JSON.stringify(result) + ")")
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return createJsonResponse(result);
}
